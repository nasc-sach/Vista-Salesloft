import json
import time
import subprocess
import tempfile
import os
from typing import Any, Dict, List, Optional, Type
from pydantic import BaseModel, Field
from crewai.tools import BaseTool


class PlaywrightExecutorToolSchema(BaseModel):
    """Input schema for PlaywrightExecutorTool."""
    playwright_automation: str = Field(
        default="",
        description="Generated Playwright automation code/JSON from Tool 1."
    )
    scenario_specification_blueprint: str = Field(
        default="{}",
        description="The Scenario Specification Blueprint JSON string from the upstream Scenario Generation Agent."
    )
    execution_profile: str = Field(
        default="Smoke",
        description="Execution profile (e.g., Smoke, Regression). Only scenarios in this profile are executed."
    )


class PlaywrightExecutorTool(BaseTool):
    """
    PlaywrightExecutorTool — Simulates Playwright execution based on the
    Scenario Specification Blueprint and returns structured execution observations.
    Since Playwright cannot run in the AAVA sandbox, this tool produces
    deterministic simulated observations from the blueprint data.
    """
    name: str = "Playwright Executor Tool"
    description: str = (
        "Executes Playwright automation according to the Scenario Specification Blueprint, "
        "Execution Profile, and Sequence. Collects and forwards execution observations."
    )
    args_schema: Type[BaseModel] = PlaywrightExecutorToolSchema

    def _run(
        self,
        playwright_automation: str = "",
        scenario_specification_blueprint: str = "{}",
        execution_profile: str = "Smoke",
        **kwargs: Any
    ) -> str:
        try:
            # ── Normalise inputs ───────────────────────────────────────────
            blueprint = scenario_specification_blueprint or {}
            if isinstance(blueprint, str):
                try:
                    blueprint = json.loads(blueprint)
                except Exception:
                    blueprint = {}

            exec_config = kwargs.get("execution_configuration") or {}
            if isinstance(exec_config, str):
                try:
                    exec_config = json.loads(exec_config)
                except Exception:
                    exec_config = {}

            auto_meta = kwargs.get("automation_metadata") or {}
            if isinstance(auto_meta, str):
                try:
                    auto_meta = json.loads(auto_meta)
                except Exception:
                    auto_meta = {}

            profile = str(execution_profile) if execution_profile else "Smoke"

            # ── Extract scenarios from blueprint using correct keys ────────
            # Agent 3 uses "Scenario Specifications" (not "scenarios")
            specs = blueprint.get("Scenario Specifications", [])
            scenario_map = {
                s.get("Scenario Identifier", ""): s
                for s in specs
                if s.get("Scenario Identifier")
            }

            # ── Determine execution sequence ───────────────────────────────
            seq = kwargs.get("execution_sequence") or []
            if isinstance(seq, str):
                try:
                    seq = json.loads(seq)
                except Exception:
                    seq = []

            # Fallback: use execution sequence from blueprint's Execution Profiles
            if not seq:
                exec_profiles = blueprint.get("Execution Profiles", {})
                seq = exec_profiles.get(profile, list(scenario_map.keys()))

            # ── Simulate execution for each scenario ───────────────────────
            overall_start    = time.time()
            execution_results  = []
            execution_timeline = []
            navigation_history = []
            interaction_history = []

            for scenario_id in seq:
                scenario = scenario_map.get(scenario_id)
                if not scenario:
                    execution_results.append({
                        "scenario_id":    scenario_id,
                        "scenario_name":  "Unknown",
                        "status":         "UNKNOWN",
                        "error":          "Scenario not found in Scenario Specification Blueprint.",
                        "retry_count":    0,
                        "duration":       0,
                        "phases":         []
                    })
                    continue

                # Only execute scenarios in the selected profile
                # Agent 3 uses "Execution Profile" (not "profile")
                scenario_profile = scenario.get("Execution Profile", "")
                if scenario_profile != profile:
                    execution_results.append({
                        "scenario_id":   scenario_id,
                        "scenario_name": scenario.get("Scenario Name", ""),
                        "status":        "SKIPPED",
                        "reason":        f"Scenario profile '{scenario_profile}' not in selected profile '{profile}'.",
                        "retry_count":   0,
                        "duration":      0,
                        "phases":        []
                    })
                    continue

                scenario_start = time.time()

                # Simulate phase execution
                phases_result = []
                for phase in scenario.get("Scenario Phases", ["Preparation", "Navigation", "Business Action", "Verification", "Cleanup"]):
                    phases_result.append({
                        "name":              phase,
                        "status":            "PASS",
                        "duration":          round(0.5 + (hash(scenario_id + phase) % 20) / 10, 2),
                        "observed_behaviour": f"{phase} phase executed for {scenario_id}",
                        "verification_status": "VERIFIED" if phase == "Verification" else "N/A"
                    })

                scenario_end  = time.time()
                scenario_duration = round(scenario_end - scenario_start + len(phases_result) * 0.5, 2)

                execution_results.append({
                    "scenario_id":   scenario_id,
                    "scenario_name": scenario.get("Scenario Name", ""),
                    "status":        "PASS",
                    "duration":      scenario_duration,
                    "retry_count":   0,
                    "phases":        phases_result,
                    "business_module": next(
                        (s.split("-")[0] for s in [scenario_id]), "Unknown"
                    ),
                    "execution_profile": profile
                })

                execution_timeline.append({
                    "scenario_id": scenario_id,
                    "start_time":  scenario_start,
                    "end_time":    scenario_end,
                    "duration":    scenario_duration,
                    "timestamp":   time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime(scenario_start))
                })

                navigation_history.append(
                    f"{scenario_id}: Login → Dashboard" if "AUTH" not in scenario_id
                    else f"{scenario_id}: Login Page"
                )
                interaction_history.append(
                    f"{scenario_id}: {scenario.get('Scenario Name', 'Execute Scenario')}"
                )

            overall_end = time.time()

            # ── Build output observation ───────────────────────────────────
            result = {
                "execution_results":   execution_results,
                "execution_timeline":  execution_timeline,
                "navigation_history":  navigation_history,
                "interaction_history": interaction_history,
                "console_summary": {
                    "info":     len(execution_results),
                    "warnings": 0,
                    "errors":   0,
                    "critical": []
                },
                "js_exception_summary": {
                    "reference_errors":    0,
                    "type_errors":         0,
                    "unhandled_exceptions": 0,
                    "promise_rejections":  0,
                    "page_crashes":        0,
                    "exception_count":     0
                },
                "network_summary": {
                    "request_count":      len(execution_results) * 5,
                    "successful_requests": len(execution_results) * 5,
                    "failed_requests":    0,
                    "redirect_count":     len(execution_results),
                    "timeout_count":      0,
                    "http_errors":        [],
                    "failed_endpoints":   []
                },
                "unknowns": blueprint.get("Unknown Areas", []),
                "execution_metadata": {
                    "execution_id":      f"exec-{int(overall_start)}",
                    "timestamp":         time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime(overall_start)),
                    "duration":          round(overall_end - overall_start, 2),
                    "execution_profile": profile,
                    "browser":           "Chromium (simulated)",
                    "browser_version":   "120.0",
                    "platform":          "AAVA Sandbox",
                    "environment":       "Simulation",
                    "viewport":          "1280x720",
                    "locale":            "en-US",
                    "timezone":          "UTC",
                    "retry_configuration": exec_config.get("max_retries", 0)
                },
                "automation_metadata": auto_meta,
                "overall_status": (
                    "PASS" if all(r["status"] == "PASS" for r in execution_results if r["status"] != "SKIPPED")
                    else "FAIL"
                )
            }

            return json.dumps(result, ensure_ascii=False)

        except Exception as e:
            return json.dumps({
                "overall_status": f"error: {str(e)}",
                "execution_results": [],
                "execution_timeline": [],
                "navigation_history": [],
                "interaction_history": [],
                "execution_metadata": {},
                "automation_metadata": {}
            }, ensure_ascii=False)
