import re
import json
from typing import Any, Type, Optional
from pydantic import BaseModel, Field
from crewai.tools import BaseTool


def extract_base_url(data: Any) -> str:
    """Extract base URL from dict blueprint or fallback to regex on string."""
    if isinstance(data, str):
        try:
            data = json.loads(data)
        except Exception:
            pass

    if isinstance(data, dict):
        app_summary = data.get("Application Summary", {})
        url = app_summary.get("Base URL") or app_summary.get("Entry Point", "")
        if url:
            return url.rstrip("/")
    # Fallback: regex on string representation
    match = re.search(r'https?://[^\s/"\']+', str(data))
    return match.group(0) if match else "process.env.BASE_URL"


def extract_scenarios(blueprint: Any) -> list:
    """
    Handles BOTH dict and markdown string inputs.
    Agent 4 passes blueprint as a dict, so we extract directly from dict keys.
    """
    scenarios = []

    if isinstance(blueprint, str):
        try:
            blueprint = json.loads(blueprint)
        except Exception:
            pass

    # ── DICT input (what Agent 4 actually sends) ──────────────────────
    if isinstance(blueprint, dict):
        specs = blueprint.get("Scenario Specifications", [])
        for spec in specs:
            scenario_id   = spec.get("Scenario Identifier", "")
            scenario_name = spec.get("Scenario Name", "")
            profile       = spec.get("Execution Profile", "Smoke")
            phases        = spec.get("Scenario Phases", [])
            hints         = spec.get("Automation Hints", [])
            verifications = spec.get("Verification", spec.get("Logical Assertions", []))

            if scenario_id:
                scenarios.append({
                    "id":            scenario_id,
                    "name":          scenario_name,
                    "profile":       profile,
                    "phases":        phases,
                    "hints":         hints,
                    "verifications": verifications
                })
        return scenarios

    # ── STRING / Markdown fallback ─────────────────────────────────────
    text = str(blueprint)
    ids   = re.findall(r'Scenario Identifier:\s*(\S+)', text)
    names = re.findall(r'Scenario Name:\s*(.+)', text)
    profs = re.findall(r'Execution Profile:\s*(\S+)', text)
    for i, sid in enumerate(ids):
        scenarios.append({
            "id":            sid.strip(),
            "name":          names[i].strip() if i < len(names) else "",
            "profile":       profs[i].strip() if i < len(profs) else "Smoke",
            "phases":        ["Preparation", "Navigation", "Business Action", "Verification", "Cleanup"],
            "hints":         [],
            "verifications": []
        })
    return scenarios


def generate_test_block(scenario: dict) -> str:
    """Generates one Playwright test block per scenario based on scenario ID."""
    sid  = scenario["id"]
    name = scenario["name"]

    # Determine if scenario needs pre-authenticated page
    needs_auth = any(k in sid for k in ["DASH", "NAV", "QL"])
    fixture    = "authenticatedPage: page" if needs_auth else "page"

    # Build assertions based on scenario ID
    if sid == "AUTH-001":
        body = """\
  await page.goto('/web/index.php/auth/login');
  await page.fill('input[name="username"]', process.env.APP_USERNAME || '');
  await page.fill('input[name="password"]', process.env.APP_PASSWORD || '');
  await page.click('button[type="submit"]');
  await expect(page).toHaveURL(/.*dashboard/);"""

    elif sid == "AUTH-002":
        body = """\
  await page.goto('/web/index.php/auth/login');
  await expect(page.locator('input[name="username"]')).toBeVisible();
  await expect(page.locator('input[name="password"]')).toBeVisible();
  await expect(page.locator('button[type="submit"]')).toBeVisible();
  await expect(page.locator('text=Forgot your password?')).toBeVisible();"""

    elif sid == "AUTH-003":
        body = """\
  await page.goto('/web/index.php/auth/login');
  await page.click('text=Forgot your password?');
  await expect(page).toHaveURL(/.*requestPasswordResetCode/);"""

    elif sid == "AUTH-004":
        body = """\
  await page.goto('/web/index.php/dashboard/index');
  await expect(page).toHaveURL(/.*login/);"""

    elif sid == "AUTH-005":
        body = """\
  await page.goto('/web/index.php/auth/login');
  await expect(page.locator('input[name="password"]')).toHaveAttribute('type', 'password');"""

    elif sid == "DASH-001":

    elif sid == "DASH-002":
        body = """\
  await expect(page.locator('text=Admin')).toBeVisible();
  await expect(page.locator('text=PIM')).toBeVisible();
  await expect(page.locator('text=Leave')).toBeVisible();
  await expect(page.locator('text=Time')).toBeVisible();
  await expect(page.locator('text=Recruitment')).toBeVisible();"""

    elif sid == "DASH-003":
        body = """\
  const searchField = page.locator('input[placeholder="Search"]');
  await expect(searchField).toBeVisible();
  await searchField.fill('test');"""

    elif sid == "DASH-004":
        body = """\
  await expect(page.locator('text=Assign Leave')).toBeVisible();
  await expect(page.locator('text=Leave List')).toBeVisible();
  await expect(page.locator('text=Timesheets')).toBeVisible();"""

    elif sid == "NAV-001":
        body = """\
  await page.click('text=Admin');
  await expect(page).toHaveURL(/.*admin/);"""

    elif sid == "NAV-002":
        body = """\
  await page.click('text=PIM');
  await expect(page).toHaveURL(/.*pim/);"""

    elif sid == "NAV-003":
        body = """\
  await page.click('text=Leave');
  await expect(page).toHaveURL(/.*leave/);"""

    elif sid == "NAV-004":
        body = """\
  await page.click('text=Time');
  await expect(page).toHaveURL(/.*time/);"""

    elif sid == "NAV-005":
        body = """\
  await page.click('text=Recruitment');
  await expect(page).toHaveURL(/.*recruitment/);"""

    elif sid == "QL-001":
        body = """\
  await page.click('text=Assign Leave');
  await expect(page).not.toHaveURL(/.*dashboard/);"""

    elif sid == "QL-002":
        body = """\
  await page.click('text=Leave List');
  await expect(page).not.toHaveURL(/.*dashboard/);"""

    elif sid == "QL-003":
        body = """\
  await page.click('text=Timesheets');
  await expect(page).not.toHaveURL(/.*dashboard/);"""

    elif sid == "EXT-001":
        body = """\
  await page.goto('/web/index.php/auth/login');
  const vendorLink = page.locator('a[href*="orangehrm.com"]');
  await expect(vendorLink).toBeVisible();"""

    else:
        body = f"  // TODO: implement assertions for {sid}"

    return f"""
test('{sid}: {name}', async ({{ {fixture} }}) => {{
{body}
}});
"""


# ─────────────────────────────────────────────────────────────────────────────
# Tool 1: Playwright Code Generator Tool
# ─────────────────────────────────────────────────────────────────────────────
class PlaywrightCodeGeneratorSchema(BaseModel):
    scenario_specification_blueprint: str = Field(
        default="{}",
        description="Validated Scenario Specification Blueprint from the upstream Scenario Generation Agent (pass as JSON string)."
    )


class PlaywrightCodeGeneratorTool(BaseTool):
    """
    PlaywrightCodeGeneratorTool — Generates executable Playwright test scripts.
    """
    name: str = "Playwright Code Generator Tool"
    description: str = (
        "Generates executable Playwright automation scripts from a given "
        "Scenario Specification Blueprint."
    )
    args_schema: Type[BaseModel] = PlaywrightCodeGeneratorSchema

    def _run(
        self,
        scenario_specification_blueprint: str = "{}",
        **kwargs: Any
    ) -> str:
        try:
            base_url  = extract_base_url(scenario_specification_blueprint)
            scenarios = extract_scenarios(scenario_specification_blueprint)

            test_blocks = "".join([generate_test_block(s) for s in scenarios])

            generated_code = f"""// ============================================================
// playwright.config.ts
// ============================================================
import {{ defineConfig }} from '@playwright/test';

export default defineConfig({{
  testDir: './tests',
  timeout: 30000,
  use: {{
    baseURL: process.env.BASE_URL || '{base_url}',
    headless: true,
    screenshot: 'only-on-failure',
    video: 'retain-on-failure'
  }}
}});

// ============================================================
// fixtures/auth.fixture.ts
// ============================================================
import {{ test as base, expect }} from '@playwright/test';

export const test = base.extend({{
  authenticatedPage: async ({{ page }}, use) => {{
    await page.goto('/web/index.php/auth/login');
    await page.fill('input[name="username"]', process.env.APP_USERNAME || '');
    await page.fill('input[name="password"]', process.env.APP_PASSWORD || '');
    await page.click('button[type="submit"]');
    await page.waitForURL('**/dashboard/**');
    await use(page);
  }}
}});

// ============================================================
// tests/scenarios.spec.ts — {len(scenarios)} Scenarios Generated
// ============================================================
import {{ test, expect }} from '../fixtures/auth.fixture';
{test_blocks}
"""

            meta = scenario_metadata if isinstance(scenario_metadata, dict) else {}

            return json.dumps({
                "playwright_automation": generated_code,
                "automation_metadata": {
                    "scenario_name": meta.get("name", "Application Automation Suite"),
                    "target_url": base_url,
                    "total_scenarios_parsed": len(scenarios),
                    "scenarios": [s["id"] for s in scenarios],
                    "framework": "Playwright TypeScript",
                    "version": "1.40.0"
                },
                "automation_version": "1.0.0",
                "generated_components": [
                    "playwright.config.ts",
                    "fixtures/auth.fixture.ts",
                    "tests/scenarios.spec.ts"
                ],
                "execution_readiness": len(scenarios) > 0,
                "generation_status": "success" if len(scenarios) > 0 else "error: no scenarios parsed"
            }, ensure_ascii=False)

        except Exception as e:
            return json.dumps({
                "generation_status": f"error: {str(e)}",
                "total_scenarios_parsed": 0,
                "scenarios": []
            }, ensure_ascii=False)
